===== .github/workflows/ci.yml
name: CI

# S1.5 — Integración continua bloqueante.
#
# Corre pruebas + analisis estatico del backend y lint/pruebas/build del
# frontend en cada push y cada pull request. Sin `continue-on-error` en
# ningun paso: un job falla en cuanto falla uno de sus pasos, y GitHub
# marca el check del PR como en rojo.
#
# "Bloqueante" tiene dos partes. Esta la resuelve: el workflow devuelve
# codigo de salida distinto de cero si algo falla. La otra parte —que
# GitHub impida hacer merge mientras el check este en rojo— se configura
# en GitHub (Settings → Branches → Branch protection rules → Require
# status checks to pass), no en un archivo del repo: hace falta push +
# permisos de administrador, fuera del alcance de una sesión local. Ver
# docs/dev/local-setup.md y .harness/progress.md (S1.5) para el detalle.

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  backend:
    name: Backend (pytest, ruff, mypy)
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend
    steps:
      - uses: actions/checkout@v7

      - name: Instalar Python
        uses: actions/setup-python@v7
        with:
          python-version: "3.12"
          cache: "pip"
          cache-dependency-path: |
            backend/requirements.txt
            backend/requirements-dev.txt

      - name: Instalar dependencias
        run: |
          pip install -r requirements.txt
          pip install -r requirements-dev.txt

      - name: ruff
        run: python -m ruff check src/dossier

      - name: mypy
        run: python -m mypy src/dossier

      - name: pytest (con cobertura)
        run: python -m pytest -q --cov=src/dossier --cov-report=term-missing

  frontend:
    name: Frontend (eslint, vitest, build)
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: frontend
    steps:
      - uses: actions/checkout@v7

      - name: Instalar Node
        uses: actions/setup-node@v7
        with:
          node-version: "20"
          cache: "npm"
          cache-dependency-path: frontend/package-lock.json

      - name: Instalar dependencias
        run: npm ci

      - name: eslint
        run: npm run lint

      - name: vitest
        run: npm test

      - name: next build
        run: npm run build

===== .github/workflows/imagenes.yml
name: Imágenes Docker

# Construye las imágenes del backend y del frontend fuera del VPS y las publica en
# GitHub Container Registry. Solo corre cuando la CI terminó en verde sobre main, y al
# final pide a Dokploy que redespliegue. Guía: docs/dev/despliegue-vps.md §7.

on:
  workflow_run:
    workflows: [CI]
    types: [completed]
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  packages: write

concurrency:
  group: imagenes-docker
  cancel-in-progress: true

jobs:
  publicar:
    name: Publicar ${{ matrix.servicio }}
    # Este job tiene packages:write y los secretos de Dokploy. Solo corre sobre código ya
    # fusionado en main de este repositorio: la CI también corre en pull requests (incluidos
    # forks), y el filtro `branches` de workflow_run compara solo el nombre de la rama.
    if: >-
      (github.event_name == 'workflow_dispatch' && github.ref == 'refs/heads/main') ||
      (github.event_name == 'workflow_run' &&
       github.event.workflow_run.conclusion == 'success' &&
       github.event.workflow_run.event == 'push' &&
       github.event.workflow_run.head_branch == 'main' &&
       github.event.workflow_run.head_repository.full_name == github.repository)
    runs-on: ubuntu-latest
    strategy:
      matrix:
        servicio: [backend, frontend]
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ github.event.workflow_run.head_sha || github.sha }}

      - name: Nombre de la imagen
        id: imagen
        # ghcr.io exige minúsculas: Xamppy/Project-Dossier -> xamppy/project-dossier
        run: echo "nombre=ghcr.io/${GITHUB_REPOSITORY,,}-${{ matrix.servicio }}" >> "$GITHUB_OUTPUT"

      - uses: docker/setup-buildx-action@8d2750c68a42422c14e847fe6c8ac0403b4cbd6f # v3.12.0

      - uses: docker/login-action@dbcb813823bdd20940b903addbd779551569679f # v4.6.0
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - uses: docker/build-push-action@c3c9e263c25d99ce0380d002d59b67737d91b0dc # v7.4.0
        with:
          context: ${{ matrix.servicio }}
          push: true
          # Relativa a propósito: frontend y API comparten dominio, así la imagen no
          # depende del dominio del servidor. El backend ignora este argumento.
          build-args: NEXT_PUBLIC_API_URL=/api
          tags: |
            ${{ steps.imagen.outputs.nombre }}:latest
            ${{ steps.imagen.outputs.nombre }}:${{ github.event.workflow_run.head_sha || github.sha }}
          cache-from: type=gha,scope=${{ matrix.servicio }}
          cache-to: type=gha,mode=max,scope=${{ matrix.servicio }}

  desplegar:
    name: Redesplegar en Dokploy
    needs: publicar
    runs-on: ubuntu-latest
    steps:
      - name: Avisar a Dokploy
        env:
          DOKPLOY_URL: ${{ secrets.DOKPLOY_URL }}
          DOKPLOY_API_KEY: ${{ secrets.DOKPLOY_API_KEY }}
          DOKPLOY_COMPOSE_ID: ${{ secrets.DOKPLOY_COMPOSE_ID }}
        run: |
          if [ -z "$DOKPLOY_URL" ] || [ -z "$DOKPLOY_API_KEY" ] || [ -z "$DOKPLOY_COMPOSE_ID" ]; then
            echo "::notice::Faltan los secretos DOKPLOY_URL, DOKPLOY_API_KEY o DOKPLOY_COMPOSE_ID: imágenes publicadas, sin redespliegue automático."
            exit 0
          fi
          curl --fail-with-body -sS -X POST "${DOKPLOY_URL%/}/api/compose.deploy" \
            -H "x-api-key: ${DOKPLOY_API_KEY}" \
            -H "Content-Type: application/json" \
            -d "{\"composeId\": \"${DOKPLOY_COMPOSE_ID}\"}"
